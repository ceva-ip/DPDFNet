cmake_minimum_required(VERSION 3.18)
project(dpdf_dprnn LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(DPDF_ENABLE_AVX2 "Build runtime-dispatched AVX2/FMA implementation" ON)
option(DPDF_SANITIZE "Enable ASan/UBSan" OFF)
option(DPDF_EXTENDED_MODEL "Enable tests for generated FC/CNN precision model" OFF)
set(DPDF_ORT_INCLUDE "" CACHE PATH "ORT 1.27 headers (optional custom operator)")
set(DPDF_GENERATED_MODEL "" CACHE FILEPATH "Generated complete model C source (optional)")
set(DPDF_TEST_WEIGHTS "" CACHE FILEPATH "Weights for generated-model contract tests")
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
if(DPDF_GENERATED_MODEL)
  add_library(dpdf_full SHARED "${DPDF_GENERATED_MODEL}" full_ops.c extended_ops.c $<TARGET_OBJECTS:dpdf_kernels>)
  target_include_directories(dpdf_full PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
  if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(dpdf_full PRIVATE -Wall -Wextra -Werror -ffp-contract=off)
    if(DPDF_ENABLE_AVX2 AND CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|AMD64|amd64")
      target_compile_definitions(dpdf_full PRIVATE DPDF_X86_DISPATCH)
    endif()
    if(DPDF_SANITIZE)
      target_compile_options(dpdf_full PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
      target_link_options(dpdf_full PRIVATE -fsanitize=address,undefined)
    endif()
  endif()
  if(NOT WIN32)
    target_link_libraries(dpdf_full PRIVATE m)
  endif()
endif()
add_library(dpdf_kernels OBJECT dpdf_dprnn.c)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(dpdf_kernels PRIVATE -Wall -Wextra -Werror -ffp-contract=off)
  if(DPDF_ENABLE_AVX2 AND CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|AMD64|amd64")
    target_sources(dpdf_kernels PRIVATE avx2.c fp16.c int8.c)
    target_compile_definitions(dpdf_kernels PRIVATE DPDF_X86_DISPATCH)
    set_source_files_properties(avx2.c PROPERTIES COMPILE_OPTIONS "-mavx2;-mfma")
    set_source_files_properties(fp16.c PROPERTIES COMPILE_OPTIONS "-mavx2;-mfma;-mf16c")
    set_source_files_properties(int8.c PROPERTIES COMPILE_OPTIONS "-mavx2;-mfma")
  endif()
endif()
if(DPDF_SANITIZE)
  target_compile_options(dpdf_kernels PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
add_library(dpdf_dprnn SHARED $<TARGET_OBJECTS:dpdf_kernels>)
if(NOT WIN32)
  target_link_libraries(dpdf_dprnn PRIVATE m)
endif()
include(CTest)
if(BUILD_TESTING)
  if(DPDF_ENABLE_AVX2 AND CMAKE_C_COMPILER_ID MATCHES "GNU|Clang" AND CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|AMD64|amd64")
    add_executable(dpdf_int8_contract int8_contract.c)
    target_compile_definitions(dpdf_int8_contract PRIVATE DPDF_X86_DISPATCH)
    target_link_libraries(dpdf_int8_contract PRIVATE dpdf_dprnn)
    if(NOT WIN32)
      target_link_libraries(dpdf_int8_contract PRIVATE m)
    endif()
    if(DPDF_SANITIZE)
      target_compile_options(dpdf_int8_contract PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    endif()
    add_test(NAME int8_contract COMMAND dpdf_int8_contract)
    set_tests_properties(int8_contract PROPERTIES TIMEOUT 30
      ENVIRONMENT "ASAN_OPTIONS=halt_on_error=1:abort_on_error=1:handle_segv=0;UBSAN_OPTIONS=halt_on_error=1")
  endif()
  if(DPDF_GENERATED_MODEL)
    add_executable(dpdf_layout_contract layout_contract.c)
    target_link_libraries(dpdf_layout_contract PRIVATE dpdf_full)
    if(NOT WIN32)
      target_link_libraries(dpdf_layout_contract PRIVATE m)
    endif()
    if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
      target_compile_options(dpdf_layout_contract PRIVATE -Wall -Wextra -Werror -ffp-contract=off)
      if(DPDF_ENABLE_AVX2 AND CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|AMD64|amd64")
        target_compile_definitions(dpdf_layout_contract PRIVATE DPDF_X86_DISPATCH)
      endif()
      if(DPDF_SANITIZE)
        target_compile_options(dpdf_layout_contract PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
      endif()
    endif()
    add_test(NAME layout_contract COMMAND dpdf_layout_contract)
    set_tests_properties(layout_contract PROPERTIES TIMEOUT 30
      ENVIRONMENT "ASAN_OPTIONS=halt_on_error=1:abort_on_error=1:handle_segv=0;UBSAN_OPTIONS=halt_on_error=1")
    if(NOT DPDF_TEST_WEIGHTS)
      if(DPDF_EXTENDED_MODEL)
        set(DPDF_TEST_WEIGHTS "${CMAKE_CURRENT_SOURCE_DIR}/../models/extended/weights.f32")
      else()
        set(DPDF_TEST_WEIGHTS "${CMAKE_CURRENT_SOURCE_DIR}/../models/full_c/weights.f32")
      endif()
    endif()
    if(DPDF_EXTENDED_MODEL)
      add_executable(dpdf_extended_contract extended_contract.c)
      target_link_libraries(dpdf_extended_contract PRIVATE dpdf_full)
      if(NOT WIN32)
        target_link_libraries(dpdf_extended_contract PRIVATE m)
      endif()
      if(DPDF_SANITIZE)
        target_compile_options(dpdf_extended_contract PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
      endif()
      add_test(NAME extended_contract COMMAND dpdf_extended_contract "${DPDF_TEST_WEIGHTS}")
      set_tests_properties(extended_contract PROPERTIES TIMEOUT 60
        ENVIRONMENT "ASAN_OPTIONS=halt_on_error=1:abort_on_error=1:handle_segv=0;UBSAN_OPTIONS=halt_on_error=1")
    endif()
    add_executable(dpdf_full_contract full_contract.c)
    target_link_libraries(dpdf_full_contract PRIVATE dpdf_full)
    if(NOT WIN32)
      target_link_libraries(dpdf_full_contract PRIVATE m)
    endif()
    if(DPDF_SANITIZE)
      target_compile_options(dpdf_full_contract PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    endif()
    add_test(NAME full_contract COMMAND dpdf_full_contract "${DPDF_TEST_WEIGHTS}")
    set_tests_properties(full_contract PROPERTIES TIMEOUT 60
      ENVIRONMENT "ASAN_OPTIONS=halt_on_error=1:abort_on_error=1:handle_segv=0;UBSAN_OPTIONS=halt_on_error=1")
  endif()
  add_executable(dpdf_contract contract_test.c)
  target_link_libraries(dpdf_contract PRIVATE dpdf_dprnn)
  if(NOT WIN32)
    target_link_libraries(dpdf_contract PRIVATE m)
  endif()
  if(DPDF_SANITIZE)
    target_compile_options(dpdf_contract PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  endif()
  add_test(NAME native_contract COMMAND dpdf_contract
    "${CMAKE_CURRENT_SOURCE_DIR}/../models/native_blocks/erb_0.f32")
  set_tests_properties(native_contract PROPERTIES TIMEOUT 60)
  if(DPDF_SANITIZE)
    set_tests_properties(native_contract PROPERTIES
      ENVIRONMENT "ASAN_OPTIONS=halt_on_error=1:abort_on_error=1:handle_segv=0;UBSAN_OPTIONS=halt_on_error=1")
  endif()
endif()
if(DPDF_ORT_INCLUDE)
  enable_language(CXX)
  add_library(dpdf_ort SHARED ort_bridge.cc $<TARGET_OBJECTS:dpdf_kernels>)
  target_include_directories(dpdf_ort PRIVATE "${DPDF_ORT_INCLUDE}")
  if(NOT WIN32)
    target_link_libraries(dpdf_ort PRIVATE m)
  endif()
endif()
